{{- /* Images written in markdown as /images/foo.jpg resolve out of assets/ and get a real WebP srcset, rather than being copied byte-for-byte out of static/. The CMS writes to assets/images and inserts /images/, so an editor uploading a phone photo still gets responsive output with no extra step. Anything we cannot resolve — an external URL, a missing file — falls through to a plain so the page still renders. */ -}} {{- $src := .Destination -}} {{- $alt := .PlainText | default .Title -}} {{- $img := "" -}} {{- if not (urls.Parse $src).Scheme -}} {{- $path := strings.TrimPrefix "/" $src -}} {{- /* A page bundle's own resource wins over the shared assets directory. */ -}} {{- with .Page.Resources.GetMatch $path -}} {{- $img = . -}} {{- else -}} {{- with resources.Get $path -}}{{- $img = . -}}{{- end -}} {{- end -}} {{- end -}} {{- with $img -}} {{- if eq .ResourceType "image" -}} {{- $opts := site.Params.image_options | default "webp q90 lanczos photo" -}} {{- $widths := slice 480 768 1200 1600 -}} {{- $srcset := slice -}} {{- range $w := $widths -}} {{- if ge $img.Width $w -}} {{- with $img.Resize (printf "%dx %s" $w $opts) -}} {{- $srcset = $srcset | append (printf "%s %dw" .RelPermalink .Width) -}} {{- end -}} {{- end -}} {{- end -}} {{- /* Narrower than the smallest step: one rendition, no srcset. */ -}} {{- $fallback := $img.Resize (printf "%dx %s" (int (math.Min $img.Width 1200)) $opts) -}} {{- else -}} {{- end -}} {{- else -}} {{- end -}}
